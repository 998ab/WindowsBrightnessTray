cmake_minimum_required(VERSION 3.20)
project(BrightnessTray LANGUAGES CXX RC)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(BrightnessTray WIN32 src/main.cpp src/display_names.cpp src/monitors.cpp src/ui.cpp src/calibration.cpp src/tray.cpp src/app.rc)
target_include_directories(BrightnessTray PRIVATE assets)
target_compile_features(BrightnessTray PRIVATE cxx_std_17)
target_compile_definitions(BrightnessTray PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(BrightnessTray PRIVATE /W4 /utf-8 /O2)
target_link_libraries(BrightnessTray PRIVATE user32 gdi32 shell32 comctl32 dxva2 dwmapi uxtheme advapi32)
enable_testing()
add_executable(localization_tests tests/localization_tests.cpp src/app.rc)
target_include_directories(localization_tests PRIVATE assets)
target_compile_features(localization_tests PRIVATE cxx_std_17)
target_compile_options(localization_tests PRIVATE /W4 /utf-8 /O2)
target_compile_definitions(localization_tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_test(NAME localization_catalogs COMMAND localization_tests)
add_executable(calibration_layout_tests tests/calibration_layout_tests.cpp)
target_compile_features(calibration_layout_tests PRIVATE cxx_std_17)
target_compile_options(calibration_layout_tests PRIVATE /W4 /utf-8 /O2)
add_test(NAME calibration_facing_edges COMMAND calibration_layout_tests)
add_executable(brightness_tests tests/brightness_tests.cpp)
target_compile_features(brightness_tests PRIVATE cxx_std_17)
target_compile_options(brightness_tests PRIVATE /W4 /utf-8 /O2)
add_test(NAME brightness_curves COMMAND brightness_tests)
add_executable(worker_tests tests/worker_tests.cpp src/monitors.cpp src/display_names.cpp)
